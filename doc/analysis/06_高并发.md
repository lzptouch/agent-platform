```bash
  --proxy-headers
```

### 4.1 独立 Skill 并行

```python
# services/agent/tools.py
import asyncio
from typing import List, Dict

async def execute_skills_parallel(skill_calls: List[Dict]) -> List[Dict]:
    """并发执行独立 Skill（无依赖时）"""
    if len(skill_calls) <= 1:
        return [await _execute_single_skill(skill_calls[0])]
    return await asyncio.gather(
        *[_execute_single_skill(sc) for sc in skill_calls],
        return_exceptions=True
    )
```

## 六、P1：SSE 限流

### 6.1 中间件

```python
ACTIVE_SSE = 0
MAX_SSE = 100

@app.middleware("http")
async def sse_limit_middleware(request: Request, call_next):
    global ACTIVE_SSE
    is_sse = "text/event-stream" in request.headers.get("accept", "")
    if is_sse and ACTIVE_SSE >= MAX_SSE:
        raise HTTPException(429, "SSE 连接数已达上限")
    if is_sse:
        ACTIVE_SSE += 1
    try:
        return await call_next(request)
    finally:
        if is_sse:
            ACTIVE_SSE -= 1
```

### 6.2 SSE 心跳

```python
@router.post("/agent/run")
async def agent_run(req: AgentRequest):
    async def event_generator():
        last_heartbeat = time.time()
        HEARTBEAT_INTERVAL = 15
        
        async for event in agent_service._stream_loop(...):
            if time.time() - last_heartbeat > HEARTBEAT_INTERVAL:
                yield "event: heartbeat\ndata: {}\n\n"
                last_heartbeat = time.time()
            yield event
    return StreamingResponse(event_generator(), media_type="text/event-stream")
```

## 七、P2：MCP stdio 池化

### 7.1 多连接池

```python
# mcp_manager.py（升级）
class MCPManager:
    def __init__(self):
        self.pools: dict[str, list[MCPClient]] = {}  # ★ 每个 server 一个连接池
    
    async def start_pool(self, server_name: str, pool_size: int = 3):
        """启动 pool_size 个 subprocess"""
        pool = []
        for _ in range(pool_size):
            client = await MCPClient.start_stdio(...)
            pool.append(client)
        self.pools[server_name] = pool
    
    async def call_tool(self, server, tool, args):
        """从池中取一个连接"""
        pool = self.pools[server]
        client = pool[self._round_robin_idx % len(pool)]
        return await client.call_tool(tool, args)
```

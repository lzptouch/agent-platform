# 第二部分 五层测试模型

## 三、五层模型详解

| 层 | 范围 | 工具 | 速度 | 触发时机 |
|----|------|------|------|---------|
| **Unit** | 纯函数 / 单 Skill / 单工具 | pytest | < 1s | 每次提交 |
| **Integration** | 单端点 + httpx | pytest + httpx | 1-5s | 每次提交 |
| **E2E** | 完整 5 链 + 前端 | Playwright + httpx | 10-60s | PR 合并 |
| **Performance** | 压测 | locust | 1-10min | 夜间 / 发布前 |
| **Fault** | 故障注入 | toxiproxy / mock | 5-30min | 手动触发 |

# tests/unit/test_skill_manager.py
import pytest
from app.services.skill_manager import SkillManager

class TestSkillManager:
    def test_list_skills(self):
        """测试列出所有 Skill"""
        mgr = SkillManager(Path("backend/skills"))
        skills = mgr.list_skills()
        assert len(skills) == 6
        assert {s["name"] for s in skills} == {
            "pdf", "data-analysis", "frontend-design",
            "marketing-materials", "ralph-loop", "skill-creator"
        }
    
    def test_get_skill(self):
        """测试获取单个 Skill"""
        mgr = SkillManager(Path("backend/skills"))
        skill = mgr.get_skill("pdf")
        assert skill is not None
        assert "PDF" in skill.description
    
    def test_get_nonexistent_skill(self):
        """测试不存在的 Skill"""
        mgr = SkillManager(Path("backend/skills"))
        assert mgr.get_skill("nonexistent") is None
```

# tests/integration/test_agent_endpoints.py
import pytest
from httpx import AsyncClient

@pytest.fixture
async def client():
    async with AsyncClient(app=app, base_url="http://test") as ac:
        yield ac

class TestAgentEndpoints:
    @pytest.mark.integration
    async def test_agent_run_sync(self, client, mock_llm):
        """测试同步 Agent"""
        response = await client.post("/api/v1/agent/run/sync", json={
            "prompt": "测试 prompt",
            "skills": ["pdf"],
            "enable_mcp": True,
            "enable_rag": True,
            "enable_web_research": False,
            "enable_reflection": False
        })
        assert response.status_code == 200
        data = response.json()
        assert data["success"]
    
    @pytest.mark.integration
    async def test_agent_run_sse(self, client, mock_llm):
        """测试 SSE 流式"""
        events = []
        async with client.stream("POST", "/api/v1/agent/run", json={...}) as resp:
            assert resp.status_code == 200
            async for line in resp.aiter_lines():
                if line.startswith("data:"):
                    events.append(json.loads(line[5:]))
        
        event_types = {e["event"] for e in events}
        assert "model_text" in event_types or "done" in event_types
```

@pytest.mark.e2e
async def test_5_chains_ui():
    """完整 5 链 UI 流程"""
    page = await browser.new_page()

    await page.click("[data-testid=enable-web]")
    await page.click("[data-testid=enable-reflection]")
    
    # 2. 选择 Skill
    await page.click("[data-testid=skill-pdf]")
    
    # 3. 发送问题
    await page.fill("[data-testid=prompt-input]", "测试问题")
    await page.click("[data-testid=send-btn]")
    
    # 4. 验证响应
    await page.wait_for_selector("[data-testid=model-text]", timeout=30000)
    text = await page.text_content("[data-testid=model-text]")

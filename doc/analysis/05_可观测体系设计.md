### 3.1 结构化日志 + request_id

```python
# utils/logger.py
import json
import logging
import uuid
from contextvars import ContextVar

request_id_var: ContextVar[str] = ContextVar('request_id', default='-')

class JsonFormatter(logging.Formatter):
    def format(self, record):
        return json.dumps({
            "ts": self.formatTime(record),
            "level": record.levelname,
            "logger": record.name,
            "msg": record.getMessage(),
            "request_id": request_id_var.get(),
            "module": record.module,
            "func": record.funcName,
        }, ensure_ascii=False)


# main.py 替换
logger = logging.getLogger("agent-platform")
handler = logging.StreamHandler()
handler.setFormatter(JsonFormatter())
logger.addHandler(handler)
```

### 3.3 OpenTelemetry 自动注入

```python
# utils/tracing.py
from opentelemetry import trace
from opentelemetry.instrumentation.fastapi import FastAPIInstrumentor
from opentelemetry.instrumentation.httpx import HTTPXClientInstrumentor

def setup_tracing(app):
    provider = trace.get_tracer_provider()
    FastAPIInstrumentor.instrument_app(app)
    HTTPXClientInstrumentor().instrument()  # DashScope / Serper
```

## 四、P1：5 链专项监控

### 4.1 Skill 调用追踪

```python
# services/skill_manager.py
from app.utils.metrics import SKILL_INVOCATION

def invoke_skill(self, name: str, args: dict):
    try:
        result = self.skills[name].invoke(args)
        SKILL_INVOCATION.labels(skill_name=name, status="success").inc()
        return result
    except Exception as e:
        SKILL_INVOCATION.labels(skill_name=name, status="failure").inc()
        raise
```

### 4.2 MCP 工具追踪

```python
# services/mcp_manager.py
from app.utils.metrics import MCP_TOOL_CALLS

async def call_tool(self, server: str, tool: str, args: dict):
    try:
        result = await self.servers[server].call_tool(tool, args)
        MCP_TOOL_CALLS.labels(server=server, tool=tool, status="success").inc()
        return result
    except Exception as e:
        MCP_TOOL_CALLS.labels(server=server, tool=tool, status="failure").inc()
        raise
```

### 4.3 Deep Research 任务追踪

```python
# services/research/research_manager.py
from app.utils.metrics import DEEP_RESEARCH_TASKS, RESEARCH_TASK_DURATION

DEEP_RESEARCH_TASKS = Counter(
    "agent_platform_deep_research_tasks_total",
    "Tasks by status",
    labelnames=["status"]
)
RESEARCH_TASK_DURATION = Histogram(
    "agent_platform_deep_research_duration_seconds",
    "Task duration",
    buckets=[10, 30, 60, 120, 300, 600]
)

def _complete_task(self, task_id, status="completed"):
    duration = time.time() - self.tasks[task_id]["started_at"]
    RESEARCH_TASK_DURATION.observe(duration)
    DEEP_RESEARCH_TASKS.labels(status=status).inc()
```

## 五、推荐可观测栈
